\begin{frame}{From variational inference to reward maximization}
  \small
  \begin{block}<1->{Expand the reverse KL for $\Tau>0$}
    \[
      \begin{aligned}
      \Tau\KL(q_\theta\|p_{\Tau})
      &=\Tau\E_{X\sim q_\theta}\!\left[
          \log q_\theta(X)-\frac{R(X)}{\Tau}+\log Z_{\Tau}
        \right]\\
      &=-\underbrace{\left(
          \E_{X\sim q_\theta}\!\left[R(X)\right]
          +\Tau\mathcal H[q_\theta]
        \right)}_{J_{\Tau}(\theta)}
        +\Tau\log Z_{\Tau}.
      \end{aligned}
    \]
  \end{block}

  \begin{alertblock}<2->{The optimization objective}
    \[
      \boxed{%
      \min_\theta\KL(q_\theta\|p_{\Tau})
      \quad\Longleftrightarrow\quad
      \max_\theta J_{\Tau}(\theta)
      =\max_\theta\left(
        \E_{q_\theta}\!\left[R(X)\right]
        +\Tau\mathcal H[q_\theta]
      \right).}
    \]
  \end{alertblock}

  \uncover<3->{%
    \begin{center}
      \textbf{Zero-temperature limit:}\qquad
      $J_0(\theta):=\displaystyle\lim_{\Tau\downarrow0}J_{\Tau}(\theta)
      =\E_{X\sim q_\theta}\!\left[R(X)\right]$.
      \\[0.10cm]
      Reward maximization is variational inference with the entropy term removed.
    \end{center}
  }
\end{frame}

\begin{frame}{Reparameterization trick for variational inference}
  \footnotesize
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}
  \begin{itemize}[<+->]
    \item Move all randomness into parameter-free noise and rewrite the VI
      objective over that noise:
      \[
        \varepsilon\sim p(\varepsilon),
        \qquad X=g_\theta(\varepsilon).
      \]
      \[
        J_{\Tau}(\theta)
        =\E_{\varepsilon}\!\left[
          R(g_\theta(\varepsilon))
          -\Tau\log q_\theta(g_\theta(\varepsilon))
        \right].
      \]
    \item Differentiate through the complete sampled objective:
      \[
        \nabla_\theta J_{\Tau}
        =\E_{\varepsilon}\!\left[
          \nabla_\theta\left(
            R(g_\theta(\varepsilon))
            -\Tau\log q_\theta(g_\theta(\varepsilon))
          \right)
        \right].
      \]
    \item For a Gaussian, use its analytic entropy:
      \[
        \boxed{\widehat L_{\mathrm{path}}
        =-\frac1N\sum_{i=1}^{N}R(g_\theta(\varepsilon_i))
         -\Tau\mathcal H[q_\theta].}
      \]
  \end{itemize}
  \uncover<4->{\scriptsize
    Requires reparameterizable $q_\theta$ and differentiable $R(x)$; usually
    lower variance than the log-derivative estimator.}
\end{frame}

\begin{frame}{Log-derivative trick for variational inference}
  \footnotesize
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}
  \begin{itemize}[<+->]
    \item Differentiate the density instead of the sampled value:
      \[
        \nabla_\theta J_{\Tau}
        =\E_{X\sim q_\theta}\!\left[
          \left(R(X)-\Tau\log q_\theta(X)-b\right)
          \nabla_\theta\log q_\theta(X)
        \right].
      \]
    \item A leave-one-out baseline keeps the estimator unbiased:
      \[
        S_i=R(X_i)-\Tau\log q_\theta(X_i),
        \qquad
        b_i=\frac{1}{N-1}\sum_{j\neq i}S_j.
      \]
    \item Use the detached learning signal in the Monte Carlo loss:
      \[
        \boxed{\widehat L_{\mathrm{score}}
        =-\frac1N\sum_{i=1}^{N}
          \stopgrad(S_i-b_i)\log q_\theta(X_i).}
      \]
  \end{itemize}
  \uncover<4->{\scriptsize
    Only $\log q_\theta(X)$ must be differentiable: $R(X)$ may be a black box,
    at the cost of typically higher variance.}
\end{frame}

\begin{frame}{Reward defines a target over trajectories}
  \small
  \only<1->{%
    \begin{tikzpicture}[remember picture,overlay]
      \node[anchor=north east,fill=white,rounded corners=1.2pt,
            inner xsep=2.8pt,inner ysep=1.8pt]
        at ([xshift=-0.16cm,yshift=-0.10cm]current page.north east)
        {\includegraphics[height=0.52cm]{assets/logos/tum_logo.png}};
    \end{tikzpicture}%
  }
  \begin{block}<1->{Unnormalized joint target}
    With temperature $\Tau>0$,
    \[
      \widetilde\pi(a_{0:T},s_{0:T+1})
      :=p(s_0)\prod_{t=0}^{T}
      p(s_{t+1}\mid s_t,a_t)
      \exp\!\left(R_{\mathrm{env}}(s_t,a_t)/\Tau\right).
    \]
  \end{block}
  \uncover<2->{%
    Marginalize the states, then normalize:
    \[
      \widetilde\pi(a_{0:T})
      :=\int\widetilde\pi(a_{0:T},s_{0:T+1})\,ds_{0:T+1},
      \qquad
      \pi(a_{0:T}):=\frac{\widetilde\pi(a_{0:T})}{\mathcal Z}.
    \]
  }
  \begin{alertblock}<3->{Why this is the target}
    $\prod_t\exp(R_{\mathrm{env}}/\Tau)=
    \exp(\sum_tR_{\mathrm{env}}/\Tau)$, so high-return trajectories receive
    exponentially more probability.
  \end{alertblock}
\end{frame}

\begin{frame}{First DPI: actions to state--action trajectories}
  \only<1->{%
    \begin{tikzpicture}[remember picture,overlay]
      \node[anchor=north east,fill=white,rounded corners=1.2pt,
            inner xsep=2.8pt,inner ysep=1.8pt]
        at ([xshift=-0.16cm,yshift=-0.10cm]current page.north east)
        {\includegraphics[height=0.52cm]{assets/logos/tum_logo.png}};
    \end{tikzpicture}%
  }
  \begin{itemize}[<+->]
    \item \textbf{Intractable marginal:}
      $\displaystyle q_\theta(a_{0:T})=\int q_\theta(a_{0:T},s_{0:T+1})\,ds_{0:T+1}$
    \item \textbf{Marginalization is a channel:}
      \[
        \boxed{
        \KL\big(q_\theta(a_{0:T})\|\pi(a_{0:T})\big)
        \leq
        \KL\big(q_\theta(a_{0:T},s_{0:T+1})\|
                \pi(a_{0:T},s_{0:T+1})\big)}
      \]
    \item Rollouts expose every $(s_t,a_t,s_{t+1})$
    \item Shared dynamics cancel in the KL ratio
  \end{itemize}
\end{frame}

\begin{frame}{Reverse KL becomes maximum-entropy RL}
  \begin{itemize}[<+->]
    \item \textbf{Joint reverse KL:}
      \[
        \Tau\KL\!\left(q_\theta(a_{0:T},s_{0:T+1})\|\pi(a_{0:T},s_{0:T+1})\right)
        =-\mathcal J_T(\theta)+\Tau\log\mathcal Z
      \]
    \item \textbf{MaxEnt RL objective:}
      \[
        \mathcal J_T(\theta)
        =\E_{q_\theta}\!\left[\sum_{t=0}^{T}
        \left[R_{\mathrm{env}}(s_t,a_t)
        -\Tau\log q_\theta(a_t\mid s_t)\right]\right]
      \]
    \item Reward: exploitation
    \item Policy entropy: exploration
    \item MaxEnt RL $=$ VI over trajectories
  \end{itemize}
\end{frame}

\begin{frame}{Infinite horizon: discount reward and entropy together}
  \footnotesize
  \only<1->{%
    \begin{tikzpicture}[remember picture,overlay]
      \node[anchor=north east,fill=white,rounded corners=1.2pt,
            inner xsep=2.8pt,inner ysep=1.8pt]

  \begin{columns}[T,onlytextwidth]
    \column{0.60\textwidth}
      \begin{block}<1->{Start with a finite cutoff $T$}
        \[
          \mathcal J_T(\theta)
          =\sum_{t=0}^{T}\E_{q_\theta}\!\left[
          R_t(s_t,a_t)-\Tau_t\log q_\theta(a_t\mid s_t)\right].
        \]
        \uncover<2->{%
          \textbf{Choose deterministic time weights}
          \[
            \boxed{\Tau_t:=\gamma^t\Tau},\qquad
            \boxed{R_t(s_t,a_t):=\gamma^tR_{\mathrm{env}}(s_t,a_t)},
            \quad \gamma\in[0,1).

        }
      \end{block}
      \uncover<3->{%

        \[
          \mathcal J_T(\theta)

\begin{frame}{Infinite horizon: now let $T\to\infty$}
  \small
  \begin{block}<1->{Infinite-horizon objective}
    Taking the cutoff $T\to\infty$ gives
    \[
      \boxed{
      \mathcal J(\theta)=\E_{q_\theta}\!\left[
      \sum_{t=0}^{\infty}\gamma^t
      \left(R_{\mathrm{env}}(s_t,a_t)
      -\Tau\log q_\theta(a_t\mid s_t)\right)
      \right]}.
    \]
  \end{block}
  \begin{alertblock}<2->{Key point}
    Discounting does not change the reward--entropy trade-off at a fixed time:
    the same factor $\gamma^t$ multiplies both terms.
  \end{alertblock}
\end{frame}

\begin{frame}{The stationary objective yields a local VI update}
  \small
  \only<1->{%
    \begin{tikzpicture}[remember picture,overlay]
      \node[anchor=north east,fill=white,rounded corners=1.2pt,
            inner xsep=2.8pt,inner ysep=1.8pt]
        at ([xshift=-0.16cm,yshift=-0.10cm]current page.north east)
        {\includegraphics[height=0.52cm]{assets/logos/tum_logo.png}};
    \end{tikzpicture}%
  }
  \begin{itemize}[<+->]
    \item \textbf{Normalized discounted state visitation:}
      \[
        d^{q_{\thetar}}(s')
        :=(1-\gamma)\sum_{t=0}^{\infty}\gamma^t
        \Pr_{q_{\thetar}}(s_t=s').
      \]
    \item \textbf{Local soft policy-improvement target:}
      \[
        \pi_Q^{q_{\thetar}}(a\mid s')
        :=\frac{\exp\!\left(Q^{q_{\thetar}}(s',a)/\Tau\right)}
        {Z_Q^{q_{\thetar}}(s')}.
      \]
    \item \textbf{Stationary local surrogate:}
      \[
        \mathcal L_{\mathrm{ME}}(\theta)
        =\frac{\Tau}{1-\gamma}\,
        \E_{s'\sim d^{q_{\thetar}}}\!\left[
        \KL\!\left(q_\theta(\cdot\mid s')\|\pi_Q^{q_{\thetar}}(\cdot\mid s')\right)
        \right].
      \]
    \item $Q^{q_{\thetar}}$ and $d^{q_{\thetar}}$ remain frozen during the actor update.
  \end{itemize}
\end{frame}

\begin{frame}{Why REPPO needs a KL trust region}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{0.62\textwidth}
      \begin{block}{TRPO monotonic-improvement bound}
        The surrogate freezes visitation under the rollout policy
        $q_{\thetar}$. For some $C>0$,
        \[
          \boxed{%
          \eta(q_\theta)
          \geq
          L_{q_{\thetar}}(q_\theta)
          -C\,D_{\mathrm{KL}}^{\max}
          \!\left(q_{\thetar}\,\middle\|\,q_\theta\right)}.
        \]
      \end{block}

      \begin{alertblock}{Consequence}
        A surrogate improvement alone is insufficient. Keeping the new policy
        close to the rollout policy controls the approximation error and yields
        the monotonic-improvement guarantee.

\begin{frame}{Variance-preserving SDE and its discretization}
  \scriptsize
  \begin{block}<1->{Continuous-time VP-SDE}
    For a noise scale $\nu>0$ and rate $\beta(t)>0$,
    \[
      dX_t=-\beta(t)X_t\,dt
        +\nu\sqrt{2\beta(t)}\,dW_t,
      \qquad B(t):=\int_0^t\beta(s)\,ds.
    \]
  \end{block}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \begin{block}<2->{Exact update over one interval}
        Define
        \[
          \alpha_k:=\exp\!\left(
          -2\int_{t_{k-1}}^{t_k}\beta(s)\,ds\right).
        \]
        Then
        \[
          X^k=\sqrt{\alpha_k}\,X^{k-1}
            +\nu\sqrt{1-\alpha_k}\,\varepsilon_k,
          \quad \varepsilon_k\sim\Normal(0,I).
        \]
      \end{block}

    \column{0.49\textwidth}
      \begin{block}<3->{Forward transition kernel}
        \[
          p(x^k\mid x^{k-1})
          =\Normal\!\left(\sqrt{\alpha_k}x^{k-1},
                           \nu^2(1-\alpha_k)I\right).
        \]
      \end{block}
      \begin{block}<4->{Learned reverse transition}
        \[
          q_\theta(x^{k-1}\mid x^k)
          =\Normal\!\left(\mu_\theta(x^k,t_k),\sigma_k^2I\right),
          \quad \sigma_k>0.
        \]
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{When has the VP-SDE added enough noise?}
  \scriptsize
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}
  \begin{columns}[T,onlytextwidth]
    \column{0.56\textwidth}
      \begin{block}<1->{Assume a Gaussian target}
        If $X_0\sim\Normal(m,\Sigma)$, the VP-SDE remains Gaussian:
        \[
          X_t\sim\Normal(m_t,\Sigma_t),
        \]
        \[
          m_t=e^{-B(t)}m,\qquad
          \Sigma_t=e^{-2B(t)}(\Sigma-\nu^2I)+\nu^2I.
        \]
      \end{block}
      \begin{block}<3->{A practical ``enough noise'' condition}
        At the terminal time, require both residuals to be negligible:
        \[
          e^{-B(T)}\lVert m\rVert\leq\epsilon_m,
          \qquad
          e^{-2B(T)}\lVert\Sigma-\nu^2I\rVert\leq\epsilon_\Sigma.
        \]

        If $B(T)=\int_0^T\beta(s)\,ds$ is sufficiently large, then
        $X_T\approx\Normal(0,\nu^2I)$: the terminal prior has forgotten the target.
      \end{alertblock}

\begin{frame}{KL chain rule: paths dominate marginals}
  \small
  \begin{block}<1->{Chain rule}
    \[
      \KL\!\left(q(x^{0:K})\|p(x^{0:K})\right)
      =\KL\!\left(q(x^0)\|p(x^0)\right)
      +\E_{q(x^0)}\!\left[
        \KL\!\left(q(x^{1:K}\mid x^0)\|p(x^{1:K}\mid x^0)\right)
      \right].
    \]
  \end{block}
  \begin{alertblock}<2->{The conditional KL is nonnegative}
    \[
      \boxed{%
      \KL\!\left(q(x^0)\|p(x^0)\right)
      \leq
      \KL\!\left(q(x^{0:K})\|p(x^{0:K})\right)}
    \]
    \centering
    Endpoint density: intractable \qquad Path transitions: tractable
  \end{alertblock}
\end{frame}

\begin{frame}{The diffusion path-space upper bound}
  \begin{itemize}[<+->]
    \item \textbf{Reverse and forward paths:}
      \[
        q_\theta(x^{0:K})=q_K(x^K)\prod_{k=1}^{K}q_\theta(x^{k-1}\mid x^k),
        \quad
        p_{\Tau}(x^{0:K})=p_{\Tau}(x^0)\prod_{k=1}^{K}p(x^k\mid x^{k-1})
      \]
    \item \textbf{DPI upper bound:}
      \[
        \boxed{\KL(q_\theta(x^0)\|p_{\Tau}(x^0))
        \leq \KL(q_\theta(x^{0:K})\|p_{\Tau}(x^{0:K}))}
      \]
    \item \textbf{Trainable loss, up to $\Tau\log Z_{\Tau}$:}
      \[
        L_{\mathrm{path}}
        =\E_{q_\theta}\!\left[
        -R(x^0)+\Tau\log q_K(x^K)
        +\Tau\sum_{k=1}^{K}\log
        \frac{q_\theta(x^{k-1}\mid x^k)}
             {p(x^k\mid x^{k-1})}\right]
      \]
  \end{itemize}
\end{frame}

\begin{frame}{Optional Langevin preconditioning and clipping}
  \begin{itemize}[<+->]
    \item \textbf{Target score:}
      \[
        g_{\Tau}(x)=\operatorname{clip}
        \left(\frac{\nabla_xR(x)}{\Tau},-10^2,10^2\right)
      \]
    \item \textbf{Preconditioned drift (Vargas et al., 2023):}
      \[
        f_\theta(k,x)=\operatorname{clip}\!\left(
        \mathrm{NN}_1(k,x)+\mathrm{NN}_2(k)\odot g_{\Tau}(x),
        -10^4,10^4\right)
      \]
    \item \textbf{Reverse score:} $s_\theta(k,x)=-x/\nu^2+f_\theta(k,x)$
    \item Detached analytic gradient
    \item Score and drift clipping
    \item Comparison axis: \texttt{use\_langevin\_preconditioning}
  \end{itemize}
\end{frame}

\begin{frame}{Starting point: the diffusion-policy marginal is intractable}
  \footnotesize

    \column{0.55\textwidth}
      \begin{block}<1->{Reverse diffusion policy}
        Start with $a_t^K\sim q_{\mathrm{prior}}=\Normal(0,\nu^2I)$ and apply

        \[
          q_\theta(a_t^{k-1}\mid a_t^k,s_t),
          \qquad k=K,\ldots,1,
        \]
        before executing $a_t^0$ in the environment.
      \end{block}
      \begin{alertblock}<2->{The executed-action likelihood is implicit}
        \[
          q_\theta(a_t^0\mid s_t)
          =\int q_{\mathrm{prior}}(a_t^K)
          \prod_{k=1}^{K}q_\theta(a_t^{k-1}\mid a_t^k,s_t)

\begin{frame}{Step 1: augment both trajectory distributions}
  \fontsize{7.2}{8.2}\selectfont
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}
  \begin{block}<1->{Reverse-policy path}
    \[
      q_\theta(a^{0:K}_{0:H},s_{0:H+1})
      =p(s_0)\prod_{t=0}^{H}\!\left[
      p(s_{t+1}\mid s_t,a_t^0)q_{\mathrm{prior}}(a_t^K)
      \prod_{k=1}^{K}q_\theta(a_t^{k-1}\mid a_t^k,s_t)\right].
    \]
  \end{block}
  \begin{block}<2->{Reward-weighted forward path}
    \[
      \pi_\theta(a^{0:K}_{0:H},s_{0:H+1})
      =\frac{p(s_0)}{\mathcal Z}\prod_{t=0}^{H}\!\left[
      p(s_{t+1}\mid s_t,a_t^0)
      e^{R_{\mathrm{env}}(s_t,a_t^0)/\Tau}
      \prod_{k=1}^{K}\pi_\theta(a_t^k\mid a_t^{k-1},s_t)\right].
    \]
  \end{block}
  \uncover<3->{%
    \begin{center}
      \textcolor{TrailDark}{reverse kernels $q_\theta$ remove noise}
      \qquad\textcolor{Slate}{$\longleftrightarrow$}\qquad
      \textcolor{BrandBlue}{forward kernels $\pi_\theta$ add noise}
    \end{center}
    Marginalizing $a^{1:K}_{0:H}$ recovers the original
    state--executed-action trajectory distributions.
  }
\end{frame}

\begin{frame}{Step 2: DPI gives a tractable full-path objective}
  \scriptsize
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}
  \begin{block}<1->{Augmenting cannot decrease the reverse KL}
    \[
      \KL\!\left(q_\theta(a^0_{0:H},s_{0:H+1})
      \,\middle\|\,\pi(a^0_{0:H},s_{0:H+1})\right)
      \leq
      \KL\!\left(q_\theta(a^{0:K}_{0:H},s_{0:H+1})
      \,\middle\|\,\pi_\theta(a^{0:K}_{0:H},s_{0:H+1})\right).
    \]
  \end{block}
  \uncover<2->{%
    The shared environment dynamics cancel, and the fixed prior contributes
    only a parameter-independent constant. Weighting environment time by
    $\gamma^t$ and taking $H\to\infty$ gives
    \[
      \boxed{
      \mathcal J_{\mathrm{DA}}(\theta)
      =\E_{q_\theta}\!\left[
      \sum_{t=0}^{\infty}\gamma^t\!\left(
      R_{\mathrm{env}}(s_t,a_t^0)
      -\Tau\sum_{k=1}^{K}\log
      \frac{q_\theta(a_t^{k-1}\mid a_t^k,s_t)}
           {\pi_\theta(a_t^k\mid a_t^{k-1},s_t)}
      \right)\right].}
    \]
  }
  \begin{alertblock}<3->{Local structure exposed by the KL}
    The environment reward occurs only when $k=1$; the reverse--forward
    log ratio occurs at every denoising step. Maximizing
    $\mathcal J_{\mathrm{DA}}$ minimizes the augmented KL up to constants.
  \end{alertblock}
\end{frame}

\begin{frame}{Step 3: derive the local critic and value equations}
  \fontsize{7.3}{8.3}\selectfont
  \setlength{\abovedisplayskip}{3pt}
  \setlength{\belowdisplayskip}{3pt}
  \uncover<1->{%
    Each denoising transition is a local decision:
    \[
      \widetilde s_{\widetilde t}=(s_t,a_t^k,k),\qquad
      \widetilde a_{\widetilde t}=a_t^{k-1},\qquad
      \widetilde R_{\mathrm{DA}}
      =\mathbf 1_{\{k=1\}}R_{\mathrm{env}}(s_t,a_t^0),
    \]
    \[
      \bar\gamma(k)=
      \begin{cases}1,&k>1,\\ \gamma,&k=1.
      \end{cases}
      \quad\text{(discount only when environment time advances).}
    \]
  }
  \begin{block}<2->{Soft Bellman equations under rollout parameters $\thetar$}
    \begin{align*}
      Q_{\mathrm{DA}}^{q_{\thetar}}(\widetilde s,a_t^{k-1})
      &=\widetilde R_{\mathrm{DA}}(\widetilde s,a_t^{k-1})
        +\bar\gamma(k)\E\!\left[
          V_{\mathrm{DA}}^{q_{\thetar}}(\widetilde s')\right],\\
      V_{\mathrm{DA}}^{q_{\thetar}}(\widetilde s)
      &=\E_{a_t^{k-1}\sim q_{\thetar}(\cdot\mid\widetilde s)}\!\left[
        Q_{\mathrm{DA}}^{q_{\thetar}}(\widetilde s,a_t^{k-1})
        -\Tau\log\frac{q_{\thetar}(a_t^{k-1}\mid a_t^k,s_t)}
        {\pi_{\thetar}(a_t^k\mid a_t^{k-1},s_t)}\right].
    \end{align*}
  \end{block}
  \uncover<3->{\centering
    The usual policy entropy is replaced by a reverse--forward transition
    log ratio.\par}
\end{frame}

\begin{frame}{Step 3 continued: derive the local actor surrogate}
  \scriptsize
  \setlength{\abovedisplayskip}{4pt}
  \setlength{\belowdisplayskip}{4pt}
  \uncover<1->{%
    Freeze the rollout parameters $\thetar$ and their discounted augmented-state
    distribution $d_{\mathrm{DA}}^{q_{\thetar}}$. The local unnormalized target is
    \[
      \rho_\theta(a^{k-1}\mid\widetilde s)
      :=\pi_\theta(a^k\mid a^{k-1},s)
        \exp\!\left(
        Q_{\mathrm{DA}}^{q_{\thetar}}(\widetilde s,a^{k-1})/\Tau\right).
    \]
  }
  \begin{block}<2->{DA-MDP policy loss}
    \[
      \boxed{
      \widehat{\mathcal L}_{\mathrm{DA}}(\theta)
      =\frac{K\Tau}{1-\gamma}
      \E_{\widetilde s\sim d_{\mathrm{DA}}^{q_{\thetar}}}\!\left[
      D_{\mathrm{KL}}^{\mathrm u}\!\left(
      q_\theta(\cdot\mid\widetilde s)
      \,\middle\|\,\rho_\theta(\cdot\mid\widetilde s)
      \right)\right].}
    \]
    $D_{\mathrm{KL}}^{\mathrm u}$ is the variational free energy with an
    unnormalized right-hand argument.
  \end{block}
  \uncover<3->{%
    \[
      \left.\nabla_\theta\widehat{\mathcal L}_{\mathrm{DA}}(\theta)
      \right|_{\theta=\thetar}
      =-\left.\nabla_\theta\mathcal J_{\mathrm{DA}}(\theta)
      \right|_{\theta=\thetar}.
    \]
    Thus the actor update uses one reverse transition at a time; trust regions
    can be imposed directly on those local kernels.
  }
\end{frame}

\begin{frame}{Step 4: the local objectives define an ordinary MDP}
  \footnotesize
  \uncover<1->{%
    \textbf{Flatten the two indices only after deriving the objectives:}
    \[
      \widetilde t(t,k)=tK+(K-k),\qquad
      \widetilde s_{\widetilde t}=(s_t,a_t^k,k),\qquad
      \widetilde a_{\widetilde t}=a_t^{k-1}.
    \]
  }
  \vspace{0.05cm}
  \uncover<2->{%
    \centering
    \renewcommand{\arraystretch}{1.42}
    \begin{tabular}{@{}p{0.12\linewidth}p{0.43\linewidth}p{0.22\linewidth}p{0.10\linewidth}@{}}
      \toprule
      \textbf{Phase} & \textbf{Successor augmented state}
        & \textbf{Environment reward} & \textbf{Discount} \\
      \midrule
      $k>1$ & $(s_t,a_t^{k-1},k-1)$ deterministically
        & $0$ & $1$ \\
      $k=1$ & $(s_{t+1},a_{t+1}^{K},K)$, where
        $s_{t+1}\sim p(\cdot\mid s_t,a_t^0)$ and
        $a_{t+1}^{K}\sim q_{\mathrm{prior}}$
        & $R_{\mathrm{env}}(s_t,a_t^0)$ & $\gamma$ \\
      \bottomrule
    \end{tabular}
    \par
  }
  \begin{alertblock}<3->{What changed?}
    Internally, the environment state is fixed while the action is denoised.
    Only $a_t^0$ advances the environment and resets the diffusion index to $K$.
  \end{alertblock}
\end{frame}

\begin{frame}{Step 5: apply standard RL to local denoising transitions}
  \small
  \begin{enumerate}[<+->]
    \item At $s_t$, draw $a_t^K\sim q_{\mathrm{prior}}$.
    \item Generate $a_t^{K-1},\ldots,a_t^0$ and store all $K$ augmented
      transitions.
    \item Execute only $a_t^0$; observe $R_{\mathrm{env}}$ and $s_{t+1}$.
    \item Compute returns or advantages with the phase-dependent
      $\bar\gamma(k)$ and update actor and critic locally.
  \end{enumerate}
  \begin{block}<5->{Three instances in the paper}
    \begin{description}
      \item[DA: PPO] PPO ratios and GAE on the augmented MDP;
        $\Tau=0$ recovers DPPO.
      \item[DA: REPPO] Reparameterized maximum-entropy update with a
        local action-value critic and reverse-kernel trust regions.
      \item[DA: WPO] Score-matching projection of the same local target.
    \end{description}
  \end{block}
\end{frame}
