\begin{frame}{Lesson 1 folder: the weighted reward landscape}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      \begin{align*}
        p_{\mathrm{ref}}(x)
        &=\sum_{i=1}^{4}w_i\,Normal(x;\mu_i,\sigma^2I),\\
        R(x)&=\log p_{\mathrm{ref}}(x),\\
        (w_1,w_2,w_3,w_4)&=(0.46,0.29,0.17,0.08),\\
        \sum_iw_i&=1.
      \end{align*}
      \begin{itemize}
        \item $w_i$: probability mass of mode $i$
        \item Gold star: global reward maximum
        \item Yellow regions: high reward
        \item Gaussian policy: one mode only
      \end{itemize}
    \column{0.42\textwidth}
      \centering
      \includegraphics[width=\linewidth]{assets/l1_start.png}
      \vspace{-0.1cm}
  \end{columns}
\end{frame}

\begin{frame}{Expected reward: the first objective}
  \begin{itemize}
    \item \textbf{Policy:} diagonal Gaussian
      \[
        q_\theta(x)=\Normal\!\left(x;\mu_\theta,
          \operatorname{diag}(\sigma_\theta^2)\right)
      \]
    \item \textbf{Objective:}
      \[
        J(\theta)=\E_{q_\theta}[R(X)],\qquad
        \theta\leftarrow\theta+\eta\nabla_\theta J(\theta)
      \]
    \item \textbf{PyTorch loss:} $L(\theta)=-J(\theta)$
    \item \textbf{Outcome:} one selected mode; vanishing variance
  \end{itemize}
\end{frame}

\begin{frame}{Reparameterization trick: move randomness outside}
  \begin{itemize}
    \item \textbf{Fixed noise:}
      \[
        \varepsilon\sim\Normal(0,I),\qquad
        X=g_\theta(\varepsilon)=\mu_\theta+\sigma_\theta\odot\varepsilon
      \]
    \item \textbf{Pathwise gradient:}
      \[
        \nabla_\theta J
        =\E_\varepsilon\!\left[
          \nabla_xR(g_\theta(\varepsilon))
          \nabla_\theta g_\theta(\varepsilon)
        \right]
      \]
    \item \textbf{Monte Carlo loss:}
      \[
        \widehat L_{\mathrm{path}}
        =-\frac1N\sum_{i=1}^N R(g_\theta(\varepsilon_i))
      \]
    \item Requires differentiable $R(x)$
  \end{itemize}
\end{frame}

\begin{frame}{Log-derivative trick: differentiate the density}
  \begin{itemize}
    \item \textbf{Score identity:}
      \[
        \nabla_\theta J
        =\E_{q_\theta}\!\left[
          R(X)\nabla_\theta\log q_\theta(X)
        \right]
      \]
    \item \textbf{Baseline:} zero expected contribution
      \[
        \E_{q_\theta}[b\nabla_\theta\log q_\theta(X)]=0
      \]
    \item \textbf{Surrogate loss:}
      \[
        \widehat L_{\mathrm{score}}
        =-\frac1N\sum_i\stopgrad(R_i-b_i)\log q_\theta(x_i)
      \]
    \item No reward gradient required
  \end{itemize}
\end{frame}

\begin{frame}{Leave-one-out baseline and estimator trade-offs}
  \begin{itemize}
    \item \textbf{Leave-one-out baseline:}
      $\displaystyle b_i=\frac{1}{N-1}\sum_{j\neq i}R(x_j)$
  \end{itemize}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \begin{block}{Reparameterization}
        \begin{itemize}
          \item \texttt{rsample}
          \item Gradient through $x$
          \item Requires $\nabla_xR$
          \item Lower variance
        \end{itemize}
      \end{block}
    \column{0.47\textwidth}
      \begin{block}{Log derivative}
        \begin{itemize}
          \item \texttt{sample}
          \item Gradient through $\log q_\theta$
          \item No $\nabla_xR$
          \item Discrete variables supported
        \end{itemize}
      \end{block}
  \end{columns}
  \begin{itemize}
    \item Same expected gradient; different computation graph
  \end{itemize}
\end{frame}

\begin{frame}{Lesson 1: what you implement}
  \begin{itemize}
    \item \texttt{L1-CostMin/gaussian\_policy\_student.ipynb}
  \end{itemize}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{A. \texttt{reparameterization\_loss}}
        \begin{itemize}
          \item \texttt{policy.distribution()}
          \item $N$ samples via \texttt{rsample}
          \item Return negative mean reward
        \end{itemize}
        \[
          \widehat{L}_{\mathrm{path}}
          =-\frac1N\sum_i R(x_i).
        \]
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{B. \texttt{log\_derivative\_loss}}
        \begin{itemize}
          \item Samples via \texttt{sample}
          \item Leave-one-out baseline $b_i$
          \item Detach $A_i=R_i-b_i$
          \item Return $-\operatorname{mean}(A_i\log q_i)$
        \end{itemize}
      \end{block}
  \end{columns}
  \begin{itemize}
    \item Run checks $\rightarrow$ train both policies $\rightarrow$ create GIF
  \end{itemize}
\end{frame}

\section{Lesson 2: KL and variational inference}

\begin{frame}[plain]
  \begin{tikzpicture}[remember picture,overlay]

      {$\color{Ice}\displaystyle \min_\theta\KL(q_\theta\|p_T)
        \Longleftrightarrow
        \max_\theta\big(\E_q[R]+T\entropy(q_\theta)\big)$};
    \node[anchor=center,text=white!75,font=\small]
      at ([xshift=0.65cm,yshift=-1.55cm]current page.center)
      {Folder: \texttt{L2-MaxEntMin/} \quad $\bullet$ \quad Reward plus entropy};
  \end{tikzpicture}
\end{frame}

\begin{frame}{The recurring problem}
  \begin{itemize}
    \item \textbf{Given:} evaluable reward $R(x)$
      \[
        p_T(x)=\frac{e^{R(x)/T}}{Z_T},\qquad
        Z_T=\int e^{R(x)/T}\,dx
      \]
    \item \textbf{Learn:} sampler $q_\theta$ for reward and diversity
      \[
        \max_\theta J_T(\theta)
        =\E_{q_\theta}[R(X)]+T\entropy(q_\theta)
      \]
    \item \textbf{Progression:} richer sampler $+$ tractable upper bound
  \end{itemize}
\end{frame}

\begin{frame}{KL divergence first}
  \begin{itemize}
    \item \textbf{Definition:}
      $\displaystyle \KL(q\|p)=\E_{X\sim q}\!\left[\log\frac{q(X)}{p(X)}\right]$
  \end{itemize}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Properties}
        \begin{itemize}
          \item $\KL(q\|p)\ge 0$
          \item $\KL(q\|p)=0 \Longleftrightarrow q=p$ a.e.
          \item Asymmetric: $\KL(q\|p)\neq\KL(p\|q)$
        \end{itemize}
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Reverse KL intuition}
        \begin{itemize}
          \item Samples come from $q_\theta$
          \item Strong penalty where $p(x)\approx0$
          \item Restricted $q_\theta$: mode seeking
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Variational inference from a reward}
  \begin{itemize}
    \item \textbf{Reward-weighted target:}
      \[
        \widetilde p_T(x)=e^{R(x)/T},\qquad
        p_T(x)=\frac{\widetilde p_T(x)}{Z_T}
      \]
    \item \textbf{Reverse KL:}
      \[
        T\KL(q_\theta\|p_T)
        =-\big(\E_q[R]+T\entropy(q_\theta)\big)+T\log Z_T
      \]
    \item \textbf{Equivalent objective:}
      \[
        \boxed{\min_\theta\KL(q_\theta\|p_T)
        \Longleftrightarrow
        \max_\theta\big(\E_q[R]+T\entropy(q_\theta)\big)}
      \]
  \end{itemize}
\end{frame}

\begin{frame}{Maximum entropy changes the sample cloud}
  \begin{itemize}
    \item \textbf{Gaussian entropy:}
      $\displaystyle \entropy(q_\theta)=\frac12\sum_{d=1}^{D}
      \log(2\pi e\,\sigma_d^2)$
    \item \textbf{Pathwise loss:}
      \[
        \widehat L_{\mathrm{path}}
        =-\frac1N\sum_iR(x_i)-T\entropy(q_\theta)
      \]
    \item \textbf{Score signal:}
      \[
        A_i=R(x_i)-T\log q_\theta(x_i)-b_i,\qquad
        \widehat L_{\mathrm{score}}
        =-\frac1N\sum_i\stopgrad(A_i)\log q_\theta(x_i)
      \]
    \item Entropy prevents variance collapse
  \end{itemize}
\end{frame}

\begin{frame}{Temperature: exploitation versus diversity}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{align*}
        J_T(\theta)&=\E_q[R]+T\entropy(q),\\
        p_T(x)&\propto e^{R(x)/T}.
      \end{align*}
      \begin{itemize}
        \item $T\downarrow0$: exploit reward maxima
        \item Moderate $T$: reward $+$ finite spread
        \item Large $T$: entropy dominates
      \end{itemize}
    \column{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth]{assets/l2_end.png}
      \vspace{-0.1cm}
      \begin{itemize}
        \item[] \scriptsize $T=0.5$: visible sample spread
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Lesson 2: what you implement}
  \begin{itemize}
    \item \texttt{L2-MaxEntMin/gaussian\_policy\_student.ipynb}
  \end{itemize}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{A. \texttt{reparameterization\_maxent\_loss}}
        \begin{itemize}
          \item States via \texttt{rsample}
          \item Mean reward
          \item Exact Gaussian entropy
          \item Return $-\overline R-T\entropy(q_\theta)$
        \end{itemize}
        \[
          \widehat{L}_{\mathrm{path}}=-\widehat J_T.
        \]
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{B. \texttt{log\_derivative\_maxent\_loss}}
        \begin{itemize}
          \item \texttt{sample} and \texttt{log\_prob}
          \item $S_i=R_i-T\log q_\theta(x_i)$
          \item Leave-one-out centering; detach
          \item Return $-\operatorname{mean}(\stopgrad(S_i-b_i)\log q_i)$
        \end{itemize}
      \end{block}
  \end{columns}
  \begin{itemize}
    \item Verify: $J_T\uparrow$, $T\KL\downarrow$, finite sample spread
  \end{itemize}
\end{frame}

  \begin{tikzpicture}[remember picture,overlay]

    \node[anchor=center,text=Ice,font=\large]
      at ([xshift=0.65cm,yshift=-0.65cm]current page.center)
      {$\color{Ice}\min_\theta\KL(q_\theta(\tau)\|p_T(\tau))
        \ \Longleftrightarrow\ \max_\theta J_{\mathrm{ME}}(\theta)$};
    \node[anchor=center,text=white!75,font=\small]
      at ([xshift=0.65cm,yshift=-1.55cm]current page.center)
      {Theory only \quad $\bullet$ \quad From reward-weighted samples to
       reward-weighted trajectories};
  \end{tikzpicture}
\end{frame}

\begin{frame}{A policy induces a distribution over trajectories}
  \begin{itemize}
    \item \textbf{Trajectory:}
      $\tau=(s_0,a_0,\ldots,a_H,s_{H+1})$
    \item \textbf{Policy-induced distribution:}
      \[
        q_\theta(\tau)=p(s_0)\prod_{t=0}^{H}
        q_\theta(a_t\mid s_t)p(s_{t+1}\mid s_t,a_t)
      \]
    \item \textbf{Reward-weighted target:}
      \[
        \widetilde p_T(\tau)=p(s_0)\prod_{t=0}^{H}
        p(s_{t+1}\mid s_t,a_t)
        e^{R_{\mathrm{env}}(s_t,a_t)/T}
      \]
    \item Same VI template; trajectories replace samples
  \end{itemize}
\end{frame}

\begin{frame}{First DPI: actions to state--action trajectories}
  \begin{itemize}
    \item \textbf{Intractable marginal:}
      $\displaystyle q_\theta(a_{0:H})=\int q_\theta(a_{0:H},s_{0:H+1})\,ds$
    \item \textbf{Marginalization is a channel:}
      \[
        \boxed{
        \KL\big(q_\theta(a_{0:H})\|p_T(a_{0:H})\big)
        \leq
        \KL\big(q_\theta(a_{0:H},s_{0:H+1})\|
                p_T(a_{0:H},s_{0:H+1})\big)}
      \]
    \item Rollouts expose every $(s_t,a_t,s_{t+1})$
    \item Shared dynamics cancel in the KL ratio
  \end{itemize}
\end{frame}

\begin{frame}{Reverse KL becomes maximum-entropy RL}
  \begin{itemize}
    \item \textbf{Joint reverse KL:}
      \[
        T\KL(q_\theta(\tau)\|p_T(\tau))
        =-J_{\mathrm{ME}}(\theta)+T\log Z_T
      \]
    \item \textbf{MaxEnt RL objective:}
      \[
        J_{\mathrm{ME}}(\theta)
        =\E_{q_\theta}\sum_{t=0}^{H}
        \left[R_{\mathrm{env}}(s_t,a_t)
        -T\log q_\theta(a_t\mid s_t)\right]
      \]
    \item Reward: exploitation
    \item Policy entropy: exploration
    \item MaxEnt RL $=$ VI over trajectories
  \end{itemize}
\end{frame}

\begin{frame}{The local variational target produces policy improvement}
  \begin{itemize}
    \item \textbf{Local Boltzmann target:}
      \[
        p^*(a\mid s)
        =\frac{\exp(Q^{q_{\theta_*}}(s,a)/T)}{Z(s)}
      \]
    \item \textbf{State-conditional VI update:}
      \[
        L_{\mathrm{ME}}(\theta)
        =T\sum_t\E_{s_t\sim q_{\theta_*}}
        \KL\!\left(q_\theta(a_t\mid s_t)\|p^*(a_t\mid s_t)\right)
      \]
    \item \textbf{Soft value:}
      $\displaystyle V^q(s)=\E_{a\sim q}[Q^q(s,a)-T\log q(a\mid s)]$
    \item Trust region: keep $\theta$ near $\theta_*$
  \end{itemize}
\end{frame}

\begin{frame}{Lesson 3 map: variational inference becomes control}
  \begin{itemize}
    \item Sample $x$ $\rightarrow$ trajectory $\tau$
    \item $q_\theta(x)$ $\rightarrow$ policy-induced $q_\theta(\tau)$
    \item Reward $R(x)$ $\rightarrow$ return $\sum_tR_{\mathrm{env}}$
    \item Boltzmann target $\rightarrow$ reward-weighted trajectories
    \item Entropy bonus $\rightarrow$ causal policy entropy
    \item Reverse KL $\rightarrow$ MaxEnt RL
    \vspace{0.3cm}
    \item \textbf{Theory only:} no new code
    \item \textbf{Bridge:} distributions $\rightarrow$ trajectories
  \end{itemize}
\end{frame}

\begin{frame}{Why move beyond a Gaussian?}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{itemize}
        \item Target: multimodal
        \item Gaussian policy: unimodal
        \item Reverse KL: mode seeking
        \item Entropy: wider mode, not more modes
      \end{itemize}
    \column{0.46\textwidth}
      \begin{block}{Desired sampler}
        \begin{itemize}
          \item Simple prior
          \item Many conditional transitions
          \item Multimodal endpoint
        \end{itemize}
        \[
          X^{K}\sim\Normal(0,I)
          \longrightarrow X^{K-1}\longrightarrow\cdots\longrightarrow X^0.
        \]
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Diffusion models: noise out, structure back in}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \begin{block}{Forward process: fixed}
        \begin{itemize}
          \item Add Gaussian noise
          \item Known transitions
        \end{itemize}
        \[
          x^0\longrightarrow x^1\longrightarrow\cdots
          \longrightarrow x^K\approx\Normal(0,I).
        \]
      \end{block}
    \column{0.49\textwidth}
      \begin{block}{Reverse process: learned}
        \begin{itemize}
          \item Start from noise
          \item Learn denoising transitions
        \end{itemize}
        \[
          x^K\longrightarrow x^{K-1}\longrightarrow\cdots
          \longrightarrow x^0\sim q_\theta.
        \]
      \end{block}
  \end{columns}
  \begin{itemize}
    \item Gaussian steps; multimodal composition
  \end{itemize}
\end{frame}

\begin{frame}{Variance-preserving SDE and its discretization}
  \begin{itemize}
    \item \textbf{VP-SDE:}
      \[
        dX_t=-\frac12\beta(t)X_t\,dt+\sqrt{\beta(t)}\,dW_t
      \]
    \item \textbf{Forward kernel:}
      $\displaystyle p(x^k\mid x^{k-1})
      =\Normal(\sqrt{\alpha_k}x^{k-1},\beta_k I)$
    \item \textbf{Reverse kernel:}
      \[
        q_\theta(x^{k-1}\mid x^k)
        =\Normal\!\left(
        \frac{x^k+\beta_k s_\theta(x^k,k)}{\sqrt{\alpha_k}},
        \beta_k I\right)
      \]
    \item $q_K=\Normal(0,I)$; $K=24$; $\beta_k:0.02\rightarrow0.18$
    \item Schedule flag: fixed or learned monotone interior points
  \end{itemize}
\end{frame}

\begin{frame}{Data-processing inequality: paths dominate marginals}
  \begin{itemize}
    \item \textbf{DPI for channel $M$:}
      $\displaystyle \KL(qM\|pM)\leq\KL(q\|p)$
    \item \textbf{KL chain rule:}
      \[
        \KL(q(x,z)\|p(x,z))
        =\KL(q(x)\|p(x))
        +\E_{q(x)}\KL(q(z\mid x)\|p(z\mid x))
      \]
    \item Marginal density $q_\theta(x^0)$: intractable
    \item Path transition densities: tractable
    \item Joint path KL: upper bound on endpoint KL
  \end{itemize}
\end{frame}

\begin{frame}{The diffusion path-space upper bound}
  \begin{itemize}
    \item \textbf{Reverse and forward paths:}
      \[
        q_\theta(x^{0:K})=q_K(x^K)\prod_{k=1}^{K}q_\theta(x^{k-1}\mid x^k),
        \quad
        p_T(x^{0:K})=p_T(x^0)\prod_{k=1}^{K}p(x^k\mid x^{k-1})
      \]
    \item \textbf{DPI upper bound:}
      \[
        \boxed{T\KL(q_\theta(x^0)\|p_T(x^0))
        \leq T\KL(q_\theta(x^{0:K})\|p_T(x^{0:K}))}
      \]
    \item \textbf{Trainable loss, up to $T\log Z_T$:}
      \[
        L_{\mathrm{path}}
        =\E_{q_\theta}\!\left[
        -R(x^0)+T\log q_K(x^K)
        +T\sum_{k=1}^{K}\log
        \frac{q_\theta(x^{k-1}\mid x^k)}
             {p(x^k\mid x^{k-1})}\right]
      \]
  \end{itemize}
\end{frame}

\begin{frame}{Optional Langevin preconditioning and clipping}
  \begin{itemize}
    \item \textbf{Target score:}
      \[
        g_T(x)=\operatorname{clip}
        \left(\frac{\nabla_xR(x)}{T},-10^2,10^2\right)
      \]
    \item \textbf{Preconditioned drift (Vargas et al., 2023):}
      \[
        f_\theta(k,x)=\operatorname{clip}\!\left(
        \mathrm{NN}_1(k,x)+\mathrm{NN}_2(k)\odot g_T(x),
        -10^4,10^4\right)
      \]
    \item \textbf{Reverse score:} $s_\theta(k,x)=-x+f_\theta(k,x)$
    \item Detached analytic gradient
    \item Score and drift clipping
    \item Flag: \texttt{USE\_LANGEVIN\_PRECONDITIONING}
  \end{itemize}
\end{frame}

\begin{frame}{Diffusion policy: one action becomes a denoising trajectory}
  \begin{itemize}
    \item \textbf{Prior action:} $a_t^K\sim q_{\mathrm{prior}}$
    \item \textbf{Conditional denoising:}
      \[
        a_t^{k-1}\sim q_\theta(a_t^{k-1}\mid a_t^k,s_t),
        \qquad k=K,\ldots,1
      \]
    \item \textbf{Executed action:} $a_t^0$
    \item \textbf{Implicit policy:}
      \[
        q_\theta(a_t^0\mid s_t)=
        \int q_{\mathrm{prior}}(a_t^K)
        \prod_{k=1}^{K}q_\theta(a_t^{k-1}\mid a_t^k,s_t)\,da_t^{1:K}
      \]
    \item Multimodal samples; intractable marginal density
  \end{itemize}
\end{frame}

\begin{frame}{Second DPI: include every diffusion latent}
  \begin{itemize}
    \item \textbf{Second DPI:}
      \[
        \KL(q_\theta(a_{0:H},s)\|p_T(a_{0:H},s))
        \le
        \KL(q_\theta(a^{0:K}_{0:H},s)\|p_T(a^{0:K}_{0:H},s))
      \]
    \item \textbf{Upper-bound decomposition:}
      \[
        \E\sum_{t=0}^{H}\sum_{k=1}^{K}
        \left[
        T\log\frac{q_\theta(a_t^{k-1}\mid a_t^k,s_t)}
        {p(a_t^k\mid a_t^{k-1},s_t)}
        -\mathbf 1_{\{k=1\}}R_{\mathrm{env}}(s_t,a_t^0)
        \right]
      \]
    \item Local, evaluable transition factors
  \end{itemize}
\end{frame}

\begin{frame}{Flatten two clocks into a diffusion-augmented MDP}
  \begin{columns}[T,onlytextwidth]
    \column{0.51\textwidth}
      \begin{itemize}
        \item \textbf{Flattened time:}
      \end{itemize}
      \begin{align*}
        \widetilde t(t,k)&=tK+(K-k),\\
        \widetilde s_{\widetilde t}&=(s_t,a_t^k,k),\\
        \widetilde a_{\widetilde t}&=a_t^{k-1}.
      \end{align*}
      \begin{itemize}
        \item \textbf{Delayed environment reward:}
      \end{itemize}
      \[
        \widetilde R(\widetilde s,\widetilde a)=
        \begin{cases}
          0,&k>1,\\
          R_{\mathrm{env}}(s_t,a_t^0),&k=1.
        \end{cases}
      \]

\begin{frame}{DA-MDP soft reward and value recursion}
  \begin{itemize}
    \item \textbf{Augmented soft reward:}
      \[
        r_{\mathrm{soft}}^{t,k}=
        \mathbf 1_{\{k=1\}}R_{\mathrm{env}}(s_t,a_t^0)
        -T\log\frac{q_\theta(a_t^{k-1}\mid a_t^k,s_t)}
        {p(a_t^k\mid a_t^{k-1},s_t)}
      \]
    \item \textbf{Soft value recursion:}
      \[
        V_{\mathrm{DA}}(\tilde s)
        =\E_{q_\theta}\!\left[
        Q_{\mathrm{DA}}(\tilde s,\tilde a)
        -T\log\frac{q_\theta(\tilde a\mid\tilde s)}
        {p(a^k\mid\tilde a,s)}\right]
      \]
    \item Forward/reverse log ratio replaces policy entropy
  \end{itemize}
\end{frame}

\begin{frame}{A local DA-MDP policy update}
  \begin{itemize}
    \item \textbf{Local target:}
      \[
        p^*(a^{k-1}\mid\tilde s)
        \propto p(a^k\mid a^{k-1},s)
        \exp(Q_{\mathrm{DA}}(s,a^{k-1})/T)
      \]
    \item \textbf{Surrogate update:}
      \[
        \E_{\tilde s\sim q_{\theta_*}}
        \KL\!\left(q_\theta(a^{k-1}\mid\tilde s)
        \,\middle\|\,p^*(a^{k-1}\mid\tilde s)\right)
      \]
    \item One reverse step per augmented state
    \item No full-chain backpropagation
    \item PPO, REPPO, or WPO variants
  \end{itemize}
\end{frame}

\begin{frame}{DA-MDP rollout and update loop}
  \begin{itemize}
    \item Sample prior noise $a_t^K$ at state $s_t$
    \item Denoise $a_t^K\rightarrow\cdots\rightarrow a_t^0$
    \item Store every augmented transition
    \item Execute $a_t^0$; observe reward and $s_{t+1}$
    \item Compute augmented returns and advantages
    \item Apply a constrained on-policy update
  \end{itemize}
  \begin{block}{Paper result}
    \begin{itemize}
      \item PPO, REPPO, and WPO variants
      \item Standard continuous-control performance
      \item Multimodal action distributions
    \end{itemize}
  \end{block}
\end{frame}
